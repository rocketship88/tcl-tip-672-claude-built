# Expression substitution: $((expr)) means [expr {expr}] wherever a $
# substitution is recognised. The body is handed to the expression parser as
# one unsubstituted word, so nothing in it is substituted twice.
# (When built with TCL_EXPR_SUBST_PARENS=1, "$(expr)" means the same thing.)

test parse-22.1 {expression substitution: token structure} testparser {
    testparser {puts $((1+2))} 0
} {- {puts $((1+2))} 2 simple puts 1 text puts 0 word {$((1+2))} 2 exprsubst {$((1+2))} 1 text (1+2) 0 {}}
test parse-22.2 {expression substitution: inside a quoted word} testparser {
    testparser {x "a$((1+2)*3)b"} 0
} {- {x "a$((1+2)*3)b"} 2 simple x 1 text x 0 word {"a$((1+2)*3)b"} 4 text a 0 exprsubst {$((1+2)*3)} 1 text (1+2)*3 0 text b 0 {}}
test parse-22.3 {expression substitution: extra parentheses are part of the body} testparser {
    testparser {$(((1+2)*3))} 0
} {- {$(((1+2)*3))} 1 word {$(((1+2)*3))} 2 exprsubst {$(((1+2)*3))} 1 text ((1+2)*3) 0 {}}
test parse-22.4 {expression substitution: Tcl_ParseVarName} testparsevarname {
    testparsevarname {$((1+(2*3)))z} 20 0

test parse-22.16 {expression substitution: no double substitution} {
    # The *value* of a variable inside the body is never substituted again.
    set cnt 0
    set s {[incr cnt]}
    list $(([string length $s])) $cnt
} {10 0}

test parse-22.33 {expression substitution: same error as [expr]} {
    set a [catch {set v $((1+))} msg1]
    set b [catch {set v [expr {(1+)}]} msg2]
    list $a $b [string equal $msg1 $msg2]
} {1 1 1}

# $((body)) must behave exactly as [expr {body}], including error and
# non-OK return codes, both when the script is compiled and when it is not.
test parse-22.40 {expression substitution == [expr {...}]: compiled and not} -body {
    set bad {}
    foreach body {
	{1+2} {2*3+1} {max(3,9)} {"a" eq "a"} {1/0} {1+} {int(7.9)}
	{[string length abc]+1} {[return -code break]} {[return -code error boom]}
	{[return -code continue]} {[return -code 7 x]}
	{1 ? "y" : "n"} {1,2} {$x*2} {$x . 1} {$nosuchvar} {[error oops]}
	{9223372036854775807+1} {"a$x"} {{braced $x}}
    } {
	set new "set x 4; set v \$(($body))"
	set ref "set x 4; set v \[expr {$body}\]"
	proc newp {} $new
	proc refp {} $ref
	foreach {kind a b} [list \
		uncompiled [list catch $new] [list catch $ref] \
		compiled   {catch newp}      {catch refp}] {
	    set ra [list [{*}$a m1] $m1]
	    set rb [list [{*}$b m2] $m2]
	    if {$ra ne $rb} {
		lappend bad [list $kind $body $ra $rb]
	    }
	}
    }
    set bad
} -result {}
